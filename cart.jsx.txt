"import React, { useState } from \"react\";
import { Link } from \"react-router-dom\";
import { useCart } from \"@/contexts/CartContext\";
import { formatPrice } from \"@/lib/api\";
import { Minus, Plus, X } from \"lucide-react\";

export default function Cart() {
  const { items, updateQty, removeItem, subtotal } = useCart();
  const [coupon, setCoupon] = useState(\"\");
  const [appliedCoupon, setAppliedCoupon] = useState(null);
  const [couponMsg, setCouponMsg] = useState(\"\");

  const applyCoupon = () => {
    if (coupon.trim().toUpperCase() === \"PANAMA10\") {
      setAppliedCoupon(\"PANAMA10\");
      setCouponMsg(\"Kupon uygulandı: %10 indirim\");
    } else {
      setAppliedCoupon(null);
      setCouponMsg(\"Geçersiz kupon kodu\");
    }
  };

  const discount = appliedCoupon === \"PANAMA10\" ? subtotal * 0.1 : 0;
  const shipping = subtotal - discount >= 2500 || subtotal === 0 ? 0 : 49;
  const vat = (subtotal - discount) * 0.18;
  const total = subtotal - discount + shipping + vat;

  if (items.length === 0) {
    return (
      <div className=\"max-w-[1600px] mx-auto px-5 sm:px-8 py-32 text-center\" data-testid=\"empty-cart\">
        <div className=\"text-[11px] tracking-[0.3em] uppercase text-zinc-500 mb-3\">Sepetiniz</div>
        <h1 className=\"font-serif text-5xl mb-6\">Sepetinizde ürün yok.</h1>
        <Link to=\"/koleksiyon\" className=\"inline-block mt-4 px-9 py-4 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase\">
          Alışverişe Başla
        </Link>
      </div>
    );
  }

  return (
    <div className=\"max-w-[1600px] mx-auto px-5 sm:px-8 py-12\" data-testid=\"cart-page\">
      <div className=\"border-b border-zinc-200 pb-8 mb-12\">
        <div className=\"text-[11px] tracking-[0.3em] uppercase text-zinc-500 mb-3\">PANAMA</div>
        <h1 className=\"font-serif text-5xl\">Sepetim ({items.length})</h1>
      </div>

      <div className=\"grid grid-cols-1 lg:grid-cols-3 gap-16\">
        <div className=\"lg:col-span-2 divide-y divide-zinc-200\">
          {items.map((it) => (
            <div key={it.key} data-testid=\"cart-item\" className=\"py-8 flex gap-6\">
              <div className=\"w-28 sm:w-32 aspect-[3/4] bg-zinc-100 flex-shrink-0 overflow-hidden\">
                {it.image && <img src={it.image} className=\"w-full h-full object-cover\" alt={it.name} />}
              </div>
              <div className=\"flex-1 flex flex-col\">
                <div className=\"flex justify-between gap-4\">
                  <div>
                    <div className=\"text-sm tracking-wide\">{it.name}</div>
                    <div className=\"text-xs text-zinc-500 mt-1\">Beden: {it.size} · Renk: {it.color || \"—\"}</div>
                  </div>
                  <button onClick={() => removeItem(it.key)} data-testid=\"remove-item\" aria-label=\"remove\">
                    <X size={16} strokeWidth={1.25} />
                  </button>
                </div>
                <div className=\"mt-auto flex items-end justify-between pt-4\">
                  <div className=\"flex items-center border border-zinc-300\">
                    <button onClick={() => updateQty(it.key, it.quantity - 1)} className=\"w-9 h-9 flex items-center justify-center\" data-testid=\"dec-qty\">
                      <Minus size={13} strokeWidth={1.25} />
                    </button>
                    <span className=\"w-9 text-center text-sm\">{it.quantity}</span>
                    <button onClick={() => updateQty(it.key, it.quantity + 1)} className=\"w-9 h-9 flex items-center justify-center\" data-testid=\"inc-qty\">
                      <Plus size={13} strokeWidth={1.25} />
                    </button>
                  </div>
                  <div className=\"text-sm\">{formatPrice(it.price * it.quantity)}</div>
                </div>
              </div>
            </div>
          ))}
        </div>

        {/* SUMMARY */}
        <aside className=\"bg-[#FAFAFA] p-8 self-start\" data-testid=\"cart-summary\">
          <h2 className=\"font-serif text-2xl mb-6\">Sipariş Özeti</h2>
          <div className=\"space-y-3 text-sm\">
            <Row label=\"Ara Toplam\" value={formatPrice(subtotal)} />
            {discount > 0 && <Row label=\"İndirim\" value={`- ${formatPrice(discount)}`} />}
            <Row label=\"KDV (%18)\" value={formatPrice(vat)} />
            <Row label=\"Kargo\" value={shipping === 0 ? \"Ücretsiz\" : formatPrice(shipping)} />
            <div className=\"border-t border-zinc-300 pt-3 mt-3\">
              <Row label=\"Genel Toplam\" value={formatPrice(total)} bold />
            </div>
          </div>

          <div className=\"mt-8\">
            <div className=\"text-[11px] tracking-[0.25em] uppercase mb-2\">Kupon Kodu</div>
            <div className=\"flex gap-2\">
              <input
                data-testid=\"coupon-input\"
                value={coupon}
                onChange={(e) => setCoupon(e.target.value)}
                placeholder=\"PANAMA10\"
                className=\"underline-input flex-1\"
              />
              <button data-testid=\"apply-coupon\" onClick={applyCoupon} className=\"px-5 py-3 border border-[#111] text-[11px] tracking-[0.25em] uppercase\">
                Uygula
              </button>
            </div>
            {couponMsg && <div className=\"text-xs mt-2 text-zinc-500\">{couponMsg}</div>}
          </div>

          <Link
            to=\"/odeme\"
            state={{ coupon: appliedCoupon }}
            data-testid=\"checkout-cta\"
            className=\"block text-center mt-8 py-4 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase hover:bg-zinc-800\"
          >
            Güvenli Ödemeye Geç
          </Link>
          <p className=\"text-[11px] text-zinc-500 mt-4 text-center tracking-wide\">
            256-bit SSL ile şifrelenmiş güvenli ödeme.
          </p>
        </aside>
      </div>
    </div>
  );
}

function Row({ label, value, bold }) {
  return (
    <div className=\"flex justify-between\">
      <span className={bold ? \"font-medium\" : \"text-zinc-600\"}>{label}</span>
      <span className={bold ? \"font-medium\" : \"\"}>{value}</span>
    </div>
  );
}

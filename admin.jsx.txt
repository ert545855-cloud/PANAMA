"import React, { useEffect, useState } from \"react\";
import { Navigate, Link } from \"react-router-dom\";
import { useAuth } from \"@/contexts/AuthContext\";
import { api, formatPrice } from \"@/lib/api\";
import { toast } from \"sonner\";
import { Trash2 } from \"lucide-react\";

const EMPTY = {
  name: \"\",
  price: 0,
  category: \"pantolon\",
  description: \"\",
  images: \"\",
  colors: \"\",
  sizes: \"XS,S,M,L,XL\",
  fabric: \"\",
  featured: false,
  new_arrival: false,
  stock: 50,
};

export default function Admin() {
  const { user, loading } = useAuth();
  const [products, setProducts] = useState([]);
  const [form, setForm] = useState(EMPTY);
  const [editingId, setEditingId] = useState(null);
  const [busy, setBusy] = useState(false);

  const refresh = () => api.get(\"/products\", { params: { limit: 200 } }).then((r) => setProducts(r.data));
  useEffect(() => { refresh(); }, []);

  if (loading) return <div className=\"text-center py-32 text-sm text-zinc-500\">Yükleniyor…</div>;
  if (!user) return <Navigate to=\"/giris\" replace />;
  if (user.role !== \"admin\") return <Navigate to=\"/hesabim\" replace />;

  const save = async (e) => {
    e.preventDefault();
    setBusy(true);
    const payload = {
      name: form.name,
      price: Number(form.price),
      description: form.description,
      category: form.category,
      images: form.images.split(\"
\").map((s) => s.trim()).filter(Boolean),
      colors: form.colors.split(\",\").map((s) => s.trim()).filter(Boolean),
      sizes: form.sizes.split(\",\").map((s) => s.trim()).filter(Boolean),
      fabric: form.fabric,
      stock: Number(form.stock) || 0,
      featured: !!form.featured,
      new_arrival: !!form.new_arrival,
    };
    try {
      if (editingId) {
        await api.put(`/products/${editingId}`, payload);
        toast.success(\"Ürün güncellendi.\");
      } else {
        await api.post(\"/products\", payload);
        toast.success(\"Ürün eklendi.\");
      }
      setForm(EMPTY);
      setEditingId(null);
      refresh();
    } catch (e) {
      toast.error(\"İşlem başarısız.\");
    } finally {
      setBusy(false);
    }
  };

  const edit = (p) => {
    setEditingId(p.id);
    setForm({
      name: p.name,
      price: p.price,
      category: p.category,
      description: p.description || \"\",
      images: (p.images || []).join(\"
\"),
      colors: (p.colors || []).join(\",\"),
      sizes: (p.sizes || []).join(\",\"),
      fabric: p.fabric || \"\",
      stock: p.stock || 0,
      featured: !!p.featured,
      new_arrival: !!p.new_arrival,
    });
    window.scrollTo({ top: 0, behavior: \"smooth\" });
  };

  const remove = async (id) => {
    if (!window.confirm(\"Ürünü silmek istediğinize emin misiniz?\")) return;
    await api.delete(`/products/${id}`);
    toast.success(\"Silindi.\");
    refresh();
  };

  return (
    <div className=\"max-w-[1400px] mx-auto px-5 sm:px-8 py-12\" data-testid=\"admin-page\">
      <div className=\"border-b border-zinc-200 pb-8 mb-12 flex justify-between items-end\">
        <div>
          <div className=\"text-[11px] tracking-[0.3em] uppercase text-zinc-500 mb-3\">Admin</div>
          <h1 className=\"font-serif text-5xl\">Ürün Yönetimi</h1>
        </div>
        <Link to=\"/\" className=\"text-[11px] tracking-[0.25em] uppercase underline\">Siteye Dön</Link>
      </div>

      <form onSubmit={save} className=\"grid grid-cols-1 md:grid-cols-2 gap-x-8 gap-y-3 mb-16 border-b border-zinc-200 pb-12\" data-testid=\"product-form\">
        <F label=\"Ürün Adı\"><input data-testid=\"admin-name\" required value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className=\"underline-input\" /></F>
        <F label=\"Fiyat (TL)\"><input data-testid=\"admin-price\" type=\"number\" required value={form.price} onChange={(e) => setForm({ ...form, price: e.target.value })} className=\"underline-input\" /></F>
        <F label=\"Kategori\">
          <select data-testid=\"admin-category\" value={form.category} onChange={(e) => setForm({ ...form, category: e.target.value })} className=\"underline-input bg-transparent\">
            <option value=\"pantolon\">Pantolon</option>
            <option value=\"elbise\">Elbise</option>
            <option value=\"etek\">Etek</option>
            <option value=\"blazer\">Blazer</option>
          </select>
        </F>
        <F label=\"Stok\"><input data-testid=\"admin-stock\" type=\"number\" value={form.stock} onChange={(e) => setForm({ ...form, stock: e.target.value })} className=\"underline-input\" /></F>
        <F label=\"Renkler (virgülle)\"><input data-testid=\"admin-colors\" value={form.colors} onChange={(e) => setForm({ ...form, colors: e.target.value })} className=\"underline-input\" placeholder=\"Siyah, Bej\" /></F>
        <F label=\"Bedenler (virgülle)\"><input data-testid=\"admin-sizes\" value={form.sizes} onChange={(e) => setForm({ ...form, sizes: e.target.value })} className=\"underline-input\" /></F>
        <F label=\"Kumaş\"><input data-testid=\"admin-fabric\" value={form.fabric} onChange={(e) => setForm({ ...form, fabric: e.target.value })} className=\"underline-input\" /></F>
        <F label=\"Açıklama\">
          <textarea data-testid=\"admin-description\" value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} className=\"underline-input min-h-[80px]\" />
        </F>
        <div className=\"md:col-span-2\">
          <div className=\"text-[11px] tracking-[0.25em] uppercase text-zinc-500 mb-1\">Görseller (her satıra bir URL)</div>
          <textarea data-testid=\"admin-images\" value={form.images} onChange={(e) => setForm({ ...form, images: e.target.value })} className=\"underline-input min-h-[100px]\" />
        </div>
        <label className=\"flex items-center gap-3 text-sm py-2\">
          <input data-testid=\"admin-featured\" type=\"checkbox\" checked={form.featured} onChange={(e) => setForm({ ...form, featured: e.target.checked })} /> Çok Satanlar
        </label>
        <label className=\"flex items-center gap-3 text-sm py-2\">
          <input data-testid=\"admin-new-arrival\" type=\"checkbox\" checked={form.new_arrival} onChange={(e) => setForm({ ...form, new_arrival: e.target.checked })} /> Yeni Gelenler
        </label>
        <div className=\"md:col-span-2 flex gap-3 mt-4\">
          <button data-testid=\"admin-save\" disabled={busy} className=\"px-9 py-4 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase disabled:opacity-50\">
            {editingId ? (busy ? \"Güncelleniyor…\" : \"Güncelle\") : busy ? \"Kaydediliyor…\" : \"Ekle\"}
          </button>
          {editingId && (
            <button type=\"button\" onClick={() => { setEditingId(null); setForm(EMPTY); }} className=\"px-9 py-4 border border-[#111] text-[11px] tracking-[0.3em] uppercase\">İptal</button>
          )}
        </div>
      </form>

      <div className=\"grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6\" data-testid=\"admin-products\">
        {products.map((p) => (
          <div key={p.id} className=\"border border-zinc-200 p-4\">
            <div className=\"aspect-[3/4] bg-zinc-100 overflow-hidden mb-3\">
              {p.images?.[0] && <img src={p.images[0]} alt={p.name} className=\"w-full h-full object-cover\" />}
            </div>
            <div className=\"text-sm\">{p.name}</div>
            <div className=\"text-xs text-zinc-500 mt-1\">{formatPrice(p.price)} · {p.category}</div>
            <div className=\"flex gap-2 mt-3\">
              <button onClick={() => edit(p)} className=\"text-[10px] tracking-[0.25em] uppercase border-b border-[#111]\">Düzenle</button>
              <button onClick={() => remove(p.id)} className=\"text-[10px] tracking-[0.25em] uppercase text-red-600 flex items-center gap-1\"><Trash2 size={12} strokeWidth={1.25} />Sil</button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}

function F({ label, children }) {
  return (
    <div className=\"py-2\">
      <div className=\"text-[11px] tracking-[0.25em] uppercase text-zinc-500 mb-1\">{label}</div>
      {children}
    </div>
  );
}
"

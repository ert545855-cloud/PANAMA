"import React, { useMemo, useState } from \"react\";
import { Link, useLocation, useNavigate } from \"react-router-dom\";
import { useCart } from \"@/contexts/CartContext\";
import { useAuth } from \"@/contexts/AuthContext\";
import { api, formatPrice } from \"@/lib/api\";
import { CreditCard, Apple, Banknote, Smartphone, Check } from \"lucide-react\";
import { toast } from \"sonner\";

export default function Checkout() {
  const { items, subtotal, clear } = useCart();
  const { user } = useAuth();
  const nav = useNavigate();
  const location = useLocation();
  const coupon = location.state?.coupon || null;

  const [form, setForm] = useState({
    full_name: user?.name || \"\",
    email: user?.email || \"\",
    phone: \"\",
    address: \"\",
    city: \"\",
    postal_code: \"\",
  });
  const [shippingMethod, setShippingMethod] = useState(\"standard\");
  const [payment, setPayment] = useState(\"credit_card\");
  const [card, setCard] = useState({ number: \"\", name: \"\", exp: \"\", cvv: \"\" });
  const [step, setStep] = useState(1); // 1: delivery, 2: shipping, 3: payment
  const [placing, setPlacing] = useState(false);
  const [order, setOrder] = useState(null);

  const discount = coupon === \"PANAMA10\" ? subtotal * 0.1 : 0;
  const shipping = useMemo(() => {
    if (subtotal - discount >= 2500) return shippingMethod === \"express\" ? 49 : 0;
    return shippingMethod === \"express\" ? 99 : 49;
  }, [subtotal, discount, shippingMethod]);
  const vat = (subtotal - discount) * 0.18;
  const total = subtotal - discount + shipping + vat;

  if (items.length === 0 && !order) {
    return (
      <div className=\"max-w-[1600px] mx-auto px-5 sm:px-8 py-32 text-center\">
        <h1 className=\"font-serif text-4xl mb-6\">Sepetiniz boş.</h1>
        <Link to=\"/koleksiyon\" className=\"px-9 py-4 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase\">Alışverişe Başla</Link>
      </div>
    );
  }

  const placeOrder = async () => {
    setPlacing(true);
    try {
      const { data } = await api.post(\"/checkout\", {
        ...form,
        shipping_method: shippingMethod,
        payment_method: payment,
        coupon,
        items: items.map((i) => ({ product_id: i.product_id, quantity: i.quantity, size: i.size, color: i.color })),
      });
      setOrder(data);
      clear();
    } catch (e) {
      toast.error(\"Sipariş alınamadı. Lütfen bilgileri kontrol edin.\");
    } finally {
      setPlacing(false);
    }
  };

  if (order) {
    return (
      <div className=\"max-w-[900px] mx-auto px-5 py-24 text-center\" data-testid=\"order-success\">
        <div className=\"w-16 h-16 mx-auto bg-[#111] text-white flex items-center justify-center\">
          <Check size={28} strokeWidth={1.25} />
        </div>
        <h1 className=\"font-serif text-5xl mt-8\">Teşekkür ederiz.</h1>
        <p className=\"text-zinc-600 mt-4\">Sipariş numaranız: <span className=\"font-medium\">{order.order_no}</span></p>
        <p className=\"text-zinc-600 mt-2\">Kısa süre içinde size ulaşacağız.</p>
        <div className=\"mt-10 text-left bg-[#FAFAFA] p-8 max-w-md mx-auto\">
          <div className=\"text-[11px] tracking-[0.25em] uppercase mb-4\">Sipariş Detayı</div>
          <div className=\"flex justify-between text-sm py-1\"><span>Ara Toplam</span><span>{formatPrice(order.subtotal)}</span></div>
          {order.discount > 0 && <div className=\"flex justify-between text-sm py-1\"><span>İndirim</span><span>- {formatPrice(order.discount)}</span></div>}
          <div className=\"flex justify-between text-sm py-1\"><span>KDV</span><span>{formatPrice(order.vat)}</span></div>
          <div className=\"flex justify-between text-sm py-1\"><span>Kargo</span><span>{order.shipping === 0 ? \"Ücretsiz\" : formatPrice(order.shipping)}</span></div>
          <div className=\"flex justify-between font-medium pt-3 border-t border-zinc-300 mt-3\"><span>Toplam</span><span>{formatPrice(order.total)}</span></div>
        </div>
        <Link to=\"/\" className=\"inline-block mt-10 px-9 py-4 border border-[#111] text-[11px] tracking-[0.3em] uppercase hover:bg-[#111] hover:text-white\">Anasayfa</Link>
      </div>
    );
  }

  const canNext1 = form.full_name && form.email && form.phone && form.address && form.city && form.postal_code;
  const canPlace = payment === \"credit_card\" ? card.number && card.name && card.exp && card.cvv : true;

  return (
    <div className=\"max-w-[1600px] mx-auto px-5 sm:px-8 py-12\" data-testid=\"checkout-page\">
      <div className=\"border-b border-zinc-200 pb-8 mb-12\">
        <div className=\"text-[11px] tracking-[0.3em] uppercase text-zinc-500 mb-3\">Güvenli Ödeme</div>
        <h1 className=\"font-serif text-5xl\">Ödeme</h1>
      </div>

      <div className=\"grid grid-cols-1 lg:grid-cols-3 gap-12\">
        <div className=\"lg:col-span-2 space-y-12\">
          {/* STEP 1 */}
          <Step n={1} title=\"Teslimat Bilgileri\" open={step >= 1} done={step > 1} onEdit={() => setStep(1)}>
            <div className=\"grid grid-cols-1 md:grid-cols-2 gap-x-6 gap-y-2\">
              <Field label=\"Ad Soyad\" value={form.full_name} onChange={(v) => setForm({ ...form, full_name: v })} testid=\"checkout-name\" />
              <Field label=\"E-posta\" type=\"email\" value={form.email} onChange={(v) => setForm({ ...form, email: v })} testid=\"checkout-email\" />
              <Field label=\"Telefon\" value={form.phone} onChange={(v) => setForm({ ...form, phone: v })} testid=\"checkout-phone\" />
              <Field label=\"Posta Kodu\" value={form.postal_code} onChange={(v) => setForm({ ...form, postal_code: v })} testid=\"checkout-postal\" />
              <div className=\"md:col-span-2\">
                <Field label=\"Adres\" value={form.address} onChange={(v) => setForm({ ...form, address: v })} testid=\"checkout-address\" />
              </div>
              <Field label=\"Şehir\" value={form.city} onChange={(v) => setForm({ ...form, city: v })} testid=\"checkout-city\" />
            </div>
            {step === 1 && (
              <button data-testid=\"step1-next\" disabled={!canNext1} onClick={() => setStep(2)} className=\"mt-8 px-9 py-4 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase disabled:opacity-30\">
                Kargo Seçimine Geç
              </button>
            )}
          </Step>

          {/* STEP 2 */}
          <Step n={2} title=\"Kargo Seçimi\" open={step >= 2} done={step > 2} onEdit={() => setStep(2)}>
            <div className=\"space-y-3\">
              {[
                { id: \"standard\", label: \"Standart Kargo\", desc: \"2-4 iş günü\", price: subtotal - discount >= 2500 ? \"Ücretsiz\" : formatPrice(49) },
                { id: \"express\", label: \"Express Kargo\", desc: \"1 iş günü (İstanbul aynı gün)\", price: subtotal - discount >= 2500 ? formatPrice(49) : formatPrice(99) },
              ].map((s) => (
                <label key={s.id} data-testid={`shipping-${s.id}`} className={`flex items-center justify-between border p-5 cursor-pointer ${shippingMethod === s.id ? \"border-[#111]\" : \"border-zinc-200\"}`}>
                  <div className=\"flex items-center gap-4\">
                    <input type=\"radio\" name=\"shipping\" checked={shippingMethod === s.id} onChange={() => setShippingMethod(s.id)} />
                    <div>
                      <div className=\"text-sm\">{s.label}</div>
                      <div className=\"text-xs text-zinc-500 mt-0.5\">{s.desc}</div>
                    </div>
                  </div>
                  <div className=\"text-sm\">{s.price}</div>
                </label>
              ))}
            </div>
            {step === 2 && (
              <button data-testid=\"step2-next\" onClick={() => setStep(3)} className=\"mt-8 px-9 py-4 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase\">
                Ödeme Bilgilerine Geç
              </button>
            )}
          </Step>

          {/* STEP 3 */}
          <Step n={3} title=\"Ödeme Bilgileri\" open={step >= 3} done={false} onEdit={() => setStep(3)}>
            <div className=\"grid grid-cols-2 sm:grid-cols-4 gap-3\">
              <PayBtn icon={<CreditCard size={16} strokeWidth={1.25} />} label=\"Kredi Kartı\" active={payment === \"credit_card\"} onClick={() => setPayment(\"credit_card\")} testid=\"pay-credit_card\" />
              <PayBtn icon={<Apple size={16} strokeWidth={1.25} />} label=\"Apple Pay\" active={payment === \"apple_pay\"} onClick={() => setPayment(\"apple_pay\")} testid=\"pay-apple_pay\" />
              <PayBtn icon={<Smartphone size={16} strokeWidth={1.25} />} label=\"Google Pay\" active={payment === \"google_pay\"} onClick={() => setPayment(\"google_pay\")} testid=\"pay-google_pay\" />
              <PayBtn icon={<Banknote size={16} strokeWidth={1.25} />} label=\"Havale/EFT\" active={payment === \"bank_transfer\"} onClick={() => setPayment(\"bank_transfer\")} testid=\"pay-bank_transfer\" />
            </div>

            {payment === \"credit_card\" && (
              <div className=\"mt-8 grid grid-cols-1 md:grid-cols-2 gap-x-6 gap-y-2\">
                <div className=\"md:col-span-2\"><Field label=\"Kart Numarası\" value={card.number} onChange={(v) => setCard({ ...card, number: v })} testid=\"card-number\" /></div>
                <Field label=\"Kart Üzerindeki İsim\" value={card.name} onChange={(v) => setCard({ ...card, name: v })} testid=\"card-name\" />
                <Field label=\"Son Kullanma (AA/YY)\" value={card.exp} onChange={(v) => setCard({ ...card, exp: v })} testid=\"card-exp\" />
                <Field label=\"CVV\" value={card.cvv} onChange={(v) => setCard({ ...card, cvv: v })} testid=\"card-cvv\" />
              </div>
            )}
            {payment === \"bank_transfer\" && (
              <div className=\"mt-6 bg-[#FAFAFA] p-5 text-sm text-zinc-600\">
                Sipariş onayından sonra IBAN bilgileri e-postanıza gönderilecektir.
              </div>
            )}
            {(payment === \"apple_pay\" || payment === \"google_pay\") && (
              <div className=\"mt-6 bg-[#FAFAFA] p-5 text-sm text-zinc-600\">
                Sipariş onayında {payment === \"apple_pay\" ? \"Apple Pay\" : \"Google Pay\"} ile ödeme tamamlanacaktır.
              </div>
            )}

            <button data-testid=\"place-order\" disabled={!canPlace || placing} onClick={placeOrder} className=\"mt-8 w-full py-4 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase disabled:opacity-40\">
              {placing ? \"İşleniyor…\" : `Siparişi Tamamla · ${formatPrice(total)}`}
            </button>
          </Step>
        </div>

        {/* SUMMARY (sticky) */}
        <aside className=\"lg:sticky lg:top-28 self-start bg-[#FAFAFA] p-8\" data-testid=\"checkout-summary\">
          <h2 className=\"font-serif text-2xl mb-6\">Sipariş Özeti</h2>
          <div className=\"divide-y divide-zinc-200 mb-6\">
            {items.map((it) => (
              <div key={it.key} className=\"flex gap-3 py-3\">
                <div className=\"w-16 aspect-[3/4] bg-white overflow-hidden flex-shrink-0\">
                  {it.image && <img src={it.image} className=\"w-full h-full object-cover\" alt=\"\" />}
                </div>
                <div className=\"flex-1 text-xs\">
                  <div className=\"truncate\">{it.name}</div>
                  <div className=\"text-zinc-500 mt-0.5\">{it.size} · {it.color || \"—\"} · x{it.quantity}</div>
                </div>
                <div className=\"text-xs\">{formatPrice(it.price * it.quantity)}</div>
              </div>
            ))}
          </div>
          <div className=\"space-y-2 text-sm\">
            <Row label=\"Ara Toplam\" value={formatPrice(subtotal)} />
            {discount > 0 && <Row label=\"İndirim (PANAMA10)\" value={`- ${formatPrice(discount)}`} />}
            <Row label=\"KDV (%18)\" value={formatPrice(vat)} />
            <Row label=\"Kargo\" value={shipping === 0 ? \"Ücretsiz\" : formatPrice(shipping)} />
            <div className=\"border-t border-zinc-300 pt-3 mt-3\"><Row label=\"Toplam\" value={formatPrice(total)} bold /></div>
          </div>
        </aside>
      </div>
    </div>
  );
}

function Step({ n, title, open, done, onEdit, children }) {
  return (
    <div className=\"border-t border-zinc-200 pt-8\" data-testid={`step-${n}`}>
      <div className=\"flex items-center justify-between mb-6\">
        <div className=\"flex items-center gap-4\">
          <span className={`w-7 h-7 flex items-center justify-center text-[11px] ${done ? \"bg-[#111] text-white\" : open ? \"bg-[#111] text-white\" : \"bg-zinc-100\"}`}>
            {done ? <Check size={13} strokeWidth={1.5} /> : n}
          </span>
          <h3 className=\"font-serif text-2xl\">{title}</h3>
        </div>
        {done && <button onClick={onEdit} className=\"text-[11px] tracking-[0.25em] uppercase underline\">Düzenle</button>}
      </div>
      {open && <div>{children}</div>}
    </div>
  );
}

function Field({ label, value, onChange, type = \"text\", testid }) {
  return (
    <div className=\"py-2\">
      <div className=\"text-[11px] tracking-[0.25em] uppercase text-zinc-500 mb-1\">{label}</div>
      <input data-testid={testid} type={type} value={value} onChange={(e) => onChange(e.target.value)} className=\"underline-input\" />
    </div>
  );
}

function PayBtn({ icon, label, active, onClick, testid }) {
  return (
    <button data-testid={testid} onClick={onClick} className={`flex flex-col items-center justify-center gap-2 py-5 border ${active ? \"border-[#111] bg-[#111] text-white\" : \"border-zinc-200 hover:border-[#111]\"}`}>
      {icon}
      <span className=\"text-[10px] tracking-[0.2em] uppercase\">{label}</span>
    </button>
  );
}

function Row({ label, value, bold }) {
  return (
    <div className=\"flex justify-between\">
      <span className={bold ? \"font-medium\" : \"text-zinc-600\"}>{label}</span>
      <span className={bold ? \"font-medium\" : \"\"}>{value}</span>
    </div>
  );
}

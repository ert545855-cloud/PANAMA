"import React, { useState } from \"react\";
import { Link } from \"react-router-dom\";
import { Instagram, Music2, Camera } from \"lucide-react\";
import { api } from \"@/lib/api\";
import { toast } from \"sonner\";

export default function Footer() {
  const [email, setEmail] = useState(\"\");
  const [sending, setSending] = useState(false);

  const submit = async (e) => {
    e.preventDefault();
    if (!email) return;
    setSending(true);
    try {
      await api.post(\"/newsletter\", { email });
      toast.success(\"PANAMA Club'a hoşgeldin.\");
      setEmail(\"\");
    } catch {
      toast.error(\"Bir hata oluştu.\");
    } finally {
      setSending(false);
    }
  };

  return (
    <footer className=\"border-t border-zinc-200 mt-24 bg-white\">
      <div className=\"max-w-[1600px] mx-auto px-5 sm:px-8 py-20 grid grid-cols-1 md:grid-cols-4 gap-12\">
        <div className=\"md:col-span-1\">
          <div className=\"font-serif text-3xl tracking-[0.35em]\">PANAMA</div>
          <p className=\"text-xs tracking-wider text-zinc-500 mt-6 leading-relaxed\">
            Zarafetin zamansız ifadesi. Premium kadın giyim koleksiyonu.
          </p>
          <div className=\"flex gap-5 mt-6\">
            <a href=\"#\" aria-label=\"instagram\" data-testid=\"social-instagram\">
              <Instagram size={18} strokeWidth={1.25} />
            </a>
            <a href=\"#\" aria-label=\"tiktok\" data-testid=\"social-tiktok\">
              <Music2 size={18} strokeWidth={1.25} />
            </a>
            <a href=\"#\" aria-label=\"pinterest\" data-testid=\"social-pinterest\">
              <Camera size={18} strokeWidth={1.25} />
            </a>
          </div>
        </div>

        <div>
          <div className=\"text-[11px] tracking-[0.25em] uppercase mb-5\">Yardım</div>
          <ul className=\"space-y-3 text-sm text-zinc-600\">
            <li><Link to=\"/sayfa/kargo-iade\">Kargo ve İade</Link></li>
            <li><Link to=\"/sayfa/beden-rehberi\">Beden Rehberi</Link></li>
            <li><Link to=\"/sayfa/iletisim\">İletişim</Link></li>
          </ul>
        </div>

        <div>
          <div className=\"text-[11px] tracking-[0.25em] uppercase mb-5\">Kurumsal</div>
          <ul className=\"space-y-3 text-sm text-zinc-600\">
            <li><Link to=\"/sayfa/hakkimizda\">PANAMA Hakkında</Link></li>
            <li><Link to=\"/sayfa/kvkk\">KVKK</Link></li>
            <li><Link to=\"/sayfa/mesafeli-satis\">Mesafeli Satış Sözleşmesi</Link></li>
          </ul>
        </div>

        <div>
          <div className=\"text-[11px] tracking-[0.25em] uppercase mb-3\">PANAMA Club</div>
          <p className=\"text-sm text-zinc-600 mb-5\">Yeni koleksiyonları ilk sen keşfet.</p>
          <form onSubmit={submit} data-testid=\"newsletter-form\">
            <input
              data-testid=\"newsletter-input\"
              type=\"email\"
              required
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              placeholder=\"E-posta adresiniz\"
              className=\"underline-input\"
            />
            <button
              data-testid=\"newsletter-submit\"
              disabled={sending}
              type=\"submit\"
              className=\"mt-5 w-full py-3 bg-[#111] text-white text-[11px] tracking-[0.3em] uppercase hover:bg-zinc-800 transition disabled:opacity-50\"
            >
              {sending ? \"Gönderiliyor…\" : \"Katıl\"}
            </button>
          </form>
        </div>
      </div>

      <div className=\"border-t border-zinc-200\">
        <div className=\"max-w-[1600px] mx-auto px-5 sm:px-8 py-6 text-[11px] tracking-widest text-zinc-500 flex flex-col md:flex-row justify-between gap-2\">
          <span>© {new Date().getFullYear()} PANAMA · Tüm hakları saklıdır.</span>
          <span>Made in İstanbul</span>
        </div>
      </div>
    </footer>
  );
}
"

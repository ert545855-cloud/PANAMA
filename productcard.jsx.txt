"import React from \"react\";
import { Link } from \"react-router-dom\";
import { Heart } from \"lucide-react\";
import { formatPrice } from \"@/lib/api\";
import { useCart } from \"@/contexts/CartContext\";
import { toast } from \"sonner\";

export default function ProductCard({ product }) {
  const { addItem } = useCart();
  const imgs = product.images || [];
  const primary = imgs[0];
  const secondary = imgs[1] || imgs[0];

  return (
    <div data-testid=\"product-card\" className=\"group\">
      <Link to={`/urun/${product.id}`} className=\"block\">
        <div className=\"product-img-wrap\">
          {primary && <img className=\"primary\" src={primary} alt={product.name} />}
          {secondary && <img className=\"secondary\" src={secondary} alt={product.name} />}
          <button
            data-testid=\"favorite-toggle\"
            onClick={(e) => {
              e.preventDefault();
              toast(\"Favorilere eklendi.\");
            }}
            className=\"absolute top-3 right-3 z-10 w-9 h-9 bg-white/85 backdrop-blur flex items-center justify-center hover:bg-white\"
            aria-label=\"favorite\"
          >
            <Heart size={15} strokeWidth={1.25} />
          </button>
        </div>
      </Link>
      <div className=\"mt-4 flex items-start justify-between gap-4\">
        <div className=\"min-w-0\">
          <Link to={`/urun/${product.id}`}>
            <div data-testid=\"product-title\" className=\"text-sm tracking-wide truncate\">{product.name}</div>
          </Link>
          <div data-testid=\"product-price\" className=\"text-sm text-zinc-700 mt-1\">{formatPrice(product.price)}</div>
          {product.colors?.length > 0 && (
            <div className=\"flex gap-1.5 mt-2\">
              {product.colors.slice(0, 4).map((c) => (
                <span
                  key={c}
                  title={c}
                  className=\"w-2.5 h-2.5 rounded-full border border-zinc-300\"
                  style={{ background: colorMap(c) }}
                />
              ))}
            </div>
          )}
        </div>
        <button
          data-testid=\"quick-add-cart\"
          onClick={(e) => {
            e.preventDefault();
            addItem(product, { size: \"M\", color: product.colors?.[0] || \"\" });
            toast.success(\"Sepete eklendi.\");
          }}
          className=\"text-[10px] tracking-[0.25em] uppercase border-b border-[#111] hover:opacity-60 transition pt-0.5\"
        >
          + Sepet
        </button>
      </div>
    </div>
  );
}

function colorMap(c) {
  const k = (c || \"\").toLowerCase();
  if (k.includes(\"siyah\")) return \"#111111\";
  if (k.includes(\"beyaz\") || k.includes(\"krem\")) return \"#f4ecdf\";
  if (k.includes(\"bej\") || k.includes(\"camel\")) return \"#c8a98a\";
  if (k.includes(\"gri\")) return \"#6b6b6b\";
  if (k.includes(\"bordo\")) return \"#5a1f24\";
  if (k.includes(\"şampanya\") || k.includes(\"sampanya\")) return \"#e9d8b8\";
  if (k.includes(\"lacivert\")) return \"#1a2747\";
  if (k.includes(\"antrasit\")) return \"#2b2b2b\";
  return \"#d4d4d4\";
}
